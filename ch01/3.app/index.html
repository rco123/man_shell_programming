system programming

리눅스 시스템 프로그래밍에서 중요한 항목들과 각 항목에 대한 샘플 프로그램을 아래에 나열하겠습니다:

1. 파일 입출력 (File I/O)

리눅스 시스템에서 파일을 읽고 쓰는 기본적인 방법을 익히는 것이 중요합니다. open(), read(), write(), close() 시스템 콜을 사용하여 파일을 다룰 수 있습니다.

샘플 코드:

#include <stdio.h>
#include <fcntl.h>
#include <unistd.h>

int main() {
    // 파일 열기
    int fd = open("sample.txt", O_WRONLY | O_CREAT, S_IRUSR | S_IWUSR);
    if (fd == -1) {
        perror("Failed to open file");
        return 1;
    }

    // 파일에 데이터 쓰기
    const char *message = "Hello, Linux!\n";
    if (write(fd, message, sizeof(message)) == -1) {
        perror("Failed to write to file");
        close(fd);
        return 1;
    }

    // 파일 닫기
    close(fd);
    return 0;
}

2. 프로세스 관리 (Process Management)

리눅스에서 프로세스를 생성하고 관리하는 것은 시스템 프로그래밍에서 중요한 부분입니다. fork(), exec(), wait() 시스템 콜을 사용하여 자식 프로세스를 생성하고 실행할 수 있습니다.

샘플 코드:

#include <stdio.h>
#include <unistd.h>
#include <sys/wait.h>

int main() {
    pid_t pid = fork();  // 프로세스 생성

    if (pid == -1) {
        perror("Fork failed");
        return 1;
    }

    if (pid == 0) {
        // 자식 프로세스
        printf("Child process\n");
        execlp("/bin/ls", "ls", NULL);  // exec()를 사용하여 새로운 프로세스 실행
    } else {
        // 부모 프로세스
        wait(NULL);  // 자식 프로세스가 종료될 때까지 기다림
        printf("Parent process\n");
    }

    return 0;
}

3. 시그널 처리 (Signal Handling)

리눅스에서 시그널은 프로세스 간의 통신 방법입니다. signal() 또는 sigaction()을 사용하여 시그널을 처리할 수 있습니다.

샘플 코드:

#include <stdio.h>
#include <signal.h>
#include <unistd.h>

void signal_handler(int sig) {
    printf("Received signal %d\n", sig);
}

int main() {
    signal(SIGINT, signal_handler);  // SIGINT 시그널 처리 설정

    while (1) {
        printf("Running... Press Ctrl+C to send SIGINT\n");
        sleep(1);
    }

    return 0;
}

4. 스레드 관리 (Thread Management)

리눅스에서 멀티스레드 프로그래밍을 위해 pthread 라이브러리를 사용하여 스레드를 생성하고 관리할 수 있습니다.

샘플 코드:

#include <stdio.h>
#include <pthread.h>

void* thread_function(void* arg) {
    printf("Thread is running\n");
    return NULL;
}

int main() {
    pthread_t thread;
    pthread_create(&thread, NULL, thread_function, NULL);
    pthread_join(thread, NULL);  // 스레드 종료를 기다림

    printf("Main thread finished\n");
    return 0;
}

5. 메모리 관리 (Memory Management)

리눅스 시스템에서는 메모리 할당과 해제를 위해 malloc(), free()를 사용하며, 가상 메모리 관리에서는 mmap()을 사용합니다.

샘플 코드:

#include <stdio.h>
#include <stdlib.h>

int main() {
    // 메모리 동적 할당
    int *ptr = (int*)malloc(sizeof(int) * 10);
    if (ptr == NULL) {
        perror("Memory allocation failed");
        return 1;
    }

    // 메모리 사용
    for (int i = 0; i < 10; i++) {
        ptr[i] = i;
        printf("%d ", ptr[i]);
    }
    printf("\n");

    // 메모리 해제
    free(ptr);
    return 0;
}

6. 네트워크 프로그래밍 (Network Programming)

소켓을 사용하여 네트워크 프로그램을 작성할 수 있습니다. socket(), bind(), listen(), accept()와 같은 시스템 콜을 사용합니다.

샘플 코드 (서버 측):

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/socket.h>
#include <arpa/inet.h>

int main() {
    int server_fd = socket(AF_INET, SOCK_STREAM, 0);
    if (server_fd == -1) {
        perror("Socket creation failed");
        return 1;
    }

    struct sockaddr_in server_addr;
    server_addr.sin_family = AF_INET;
    server_addr.sin_port = htons(8080);
    server_addr.sin_addr.s_addr = INADDR_ANY;

    if (bind(server_fd, (struct sockaddr*)&server_addr, sizeof(server_addr)) == -1) {
        perror("Bind failed");
        close(server_fd);
        return 1;
    }

    if (listen(server_fd, 3) == -1) {
        perror("Listen failed");
        close(server_fd);
        return 1;
    }

    printf("Waiting for connections...\n");

    int new_socket = accept(server_fd, NULL, NULL);
    if (new_socket == -1) {
        perror("Accept failed");
        close(server_fd);
        return 1;
    }

    printf("Connection accepted\n");
    close(new_socket);
    close(server_fd);
    return 0;
}

7. 파일 시스템 (File System)

리눅스 시스템에서는 파일 시스템을 직접 조작할 수 있습니다. opendir(), readdir(), closedir()를 사용하여 디렉토리 내용을 탐색할 수 있습니다.

샘플 코드:

#include <stdio.h>
#include <dirent.h>

int main() {
    DIR *dir = opendir(".");
    if (dir == NULL) {
        perror("opendir failed");
        return 1;
    }

    struct dirent *entry;
    while ((entry = readdir(dir)) != NULL) {
        printf("Found: %s\n", entry->d_name);
    }

    closedir(dir);
    return 0;
}

8. 프로세스 간 통신 (IPC: Inter-Process Communication)

리눅스에서는 여러 가지 방법을 사용하여 프로세스 간 통신(IPC, Inter-Process Communication)을 수행할 수 있습니다. 대표적인 IPC 기법에는 파이프(PIPE), 메시지 큐(Message Queue), 공유 메모리(Shared Memory), 소켓(Socket), 시그널(Signal) 등이 있습니다.

1) 익명 파이프 (Anonymous Pipe)** - **익명 파이프**는 pipe() 시스템 콜을 사용하여 부모와 자식 프로세스 간에 데이터를 전송하는 방식입니다. - 부모 프로세스가 데이터를 쓰면 자식 프로세스가 이를 읽을 수 있습니다.

샘플 코드 (익명 파이프) 

#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>

int main() {
    int fd[2];
    pid_t pid;
    char buffer[100];

    if (pipe(fd) == -1) {  // 파이프 생성
        perror("pipe failed");
        return 1;
    }

    pid = fork();
    if (pid == -1) {
        perror("fork failed");
        return 1;
    }

    if (pid == 0) {  // 자식 프로세스
        close(fd[1]);  // 쓰기 닫기
        read(fd[0], buffer, sizeof(buffer));  // 읽기
        printf("Child received: %s\n", buffer);
        close(fd[0]);
    } else {  // 부모 프로세스
        close(fd[0]);  // 읽기 닫기
        write(fd[1], "Hello from parent", 18);  // 데이터 쓰기
        close(fd[1]);
    }

    return 0;
}
출력 예시 
Child received: Hello from parent

2) 명명된 파이프 (Named Pipe, FIFO) - **익명 파이프**는 부모-자식 간 통신만 가능하지만, **명명된 파이프(FIFO)**는 **서로 다른 프로세스 간**에도 통신할 수 있습니다.

샘플 코드 (FIFO 생성 및 사용) 

#include <stdio.h>
#include <stdlib.h>
#include <fcntl.h>
#include <sys/stat.h>
#include <unistd.h>

#define FIFO_NAME "mypipe"

int main() {
    mkfifo(FIFO_NAME, 0666);  // FIFO 파일 생성

    pid_t pid = fork();

    if (pid == 0) {  // 자식 프로세스: 읽기
        char buffer[100];
        int fd = open(FIFO_NAME, O_RDONLY);
        read(fd, buffer, sizeof(buffer));
        printf("Child received: %s\n", buffer);
        close(fd);
    } else {  // 부모 프로세스: 쓰기
        int fd = open(FIFO_NAME, O_WRONLY);
        write(fd, "Hello from parent", 18);
        close(fd);
    }

    return 0;
}

출력 예시 

Child received: Hello from parent

3) 메시지 큐 (Message Queue) - **메시지 큐(Message Queue)**는 **커널에서 관리하는 메시지 저장 공간**을 사용하여 프로세스 간 데이터를 주고받는 방식입니다.

샘플 코드 (메시지 큐) 

#include <stdio.h>
#include <stdlib.h>
#include <sys/ipc.h>
#include <sys/msg.h>
#include <string.h>

struct message {
    long msg_type;
    char msg_text[100];
};

int main() {
    key_t key = ftok("msgqueue", 65);
    int msgid = msgget(key, 0666 | IPC_CREAT);

    struct message msg;
    msg.msg_type = 1;
    strcpy(msg.msg_text, "Hello from process");

    msgsnd(msgid, &msg, sizeof(msg), 0);
    printf("Message sent: %s\n", msg.msg_text);

    return 0;
}
출력 예시 
Message sent: Hello from process

4) 공유 메모리 (Shared Memory) - **공유 메모리(Shared Memory)**는 **여러 프로세스가 같은 메모리 영역을 공유하여 데이터를 주고받는 방식**입니다. - shmget(), shmat(), shmdt() 등을 사용하여 공유 메모리를 생성하고 사용합니다.

샘플 코드 (공유 메모리)

#include <stdio.h>
#include <stdlib.h>
#include <sys/ipc.h>
#include <sys/shm.h>
#include <string.h>

int main() {
    key_t key = ftok("shmfile", 65);
    int shmid = shmget(key, 1024, 0666 | IPC_CREAT);

    char *str = (char*) shmat(shmid, (void*)0, 0);
    strcpy(str, "Hello from shared memory");

    printf("Data written in memory: %s\n", str);

    shmdt(str);
    return 0;
}
출력 예시 
Data written in memory: Hello from shared memory

5) 소켓을 이용한 프로세스 간 통신 - 소켓(Socket)**은 **네트워크 상에서 프로세스 간 데이터를 주고받을 때 사용**됩니다. - **로컬 프로세스 간 또는 **다른 컴퓨터와의 통신**도 가능.

샘플 코드 (TCP 서버) 

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <sys/socket.h>
#include <arpa/inet.h>

int main() {
    int server_fd = socket(AF_INET, SOCK_STREAM, 0);
    struct sockaddr_in server_addr;

    server_addr.sin_family = AF_INET;
    server_addr.sin_port = htons(8080);
    server_addr.sin_addr.s_addr = INADDR_ANY;

    bind(server_fd, (struct sockaddr*)&server_addr, sizeof(server_addr));
    listen(server_fd, 3);

    printf("Waiting for connections...\n");

    int new_socket = accept(server_fd, NULL, NULL);
    if (new_socket >= 0) {
        printf("Connection accepted\n");
        send(new_socket, "Hello from server", 18, 0);
    }

    close(new_socket);
    close(server_fd);
    return 0;
}
출력 예시 
Waiting for connections...
Connection accepted

9. 결론

리눅스에서 **프로세스 간 통신(IPC)**을 수행하는 주요 방식은 다음과 같습니다:

IPC 기법 특징 사용 예시
파이프 (PIPE) 부모-자식 간 단순 통신 pipe()
FIFO (Named Pipe) 서로 다른 프로세스 간 통신 mkfifo()
메시지 큐 (Message Queue) 메시지 기반의 비동기 통신 msgget(), msgsnd(), msgrcv()
공유 메모리 (Shared Memory) 같은 메모리 공간을 공유 shmget(), shmat(), shmdt()
소켓 (Socket) 네트워크 기반 통신 socket(), bind(), listen(), accept()

각 IPC 기법은 상황에 따라 적절하게 사용해야 합니다. - 빠른 데이터 공유가 필요하면 공유 메모리, - 네트워크나 원격 프로세스 간 통신이 필요하면 소켓, - 부모-자식 간 단순 데이터 전송이라면 **파이프**가 적합합니다. 🚀

리눅스 시스템 프로그래밍을 할 때, IPC 기법을 적절히 활용하면 강력한 멀티프로세싱 애플리케이션을 개발할 수 있습니다!
